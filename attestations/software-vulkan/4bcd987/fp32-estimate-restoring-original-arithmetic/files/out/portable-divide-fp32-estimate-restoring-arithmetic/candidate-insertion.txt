#if defined(SIRIUS_PB32_DIVIDE_FP32_ESTIMATE)
    // This normal binary32 proposal is untrusted; only exact integer checks
    // below can authorize an early return. The original fallback is unchanged.
    const uint numeratorExponentBits = aa.significand < bb.significand ? 0x4d000000u : 0x4c800000u;
    const uint numeratorBits = numeratorExponentBits | (aa.significand & 0x007fffffu);
    const uint denominatorBits = 0x3f800000u | (bb.significand & 0x007fffffu);
#if defined(__cplusplus)
    static_assert(sizeof(float) == sizeof(uint));
    const float proposal = std::bit_cast<float>(numeratorBits) / std::bit_cast<float>(denominatorBits);
    const uint proposalBits = std::bit_cast<uint>(proposal);
#else
    const float proposal = asfloat(numeratorBits) / asfloat(denominatorBits);
    const uint proposalBits = asuint(proposal);
#endif
    // Reject sign, non-normal and out-of-binade results before bit extraction.
    if ((proposalBits & 0xff800000u) == 0x4c800000u) {
        const uint h = ((proposalBits & 0x007fffffu) | 0x00800000u) << 3;
        const uint guess = h - 16u;
        const PB32Wide product = PB32WideProduct(guess, bb.significand);
        const uint numeratorLo = remainder << 26;
        const uint numeratorHi = remainder >> 6;
        // Unsigned two-word subtraction is modulo 2^64, without a >= b
        // precondition. Since both magnitudes are <2^51, a negative result
        // cannot have high word zero and therefore cannot pass this guard.
        const uint residualLo = numeratorLo - product.lo;
        const uint residualHi = numeratorHi - product.hi - (numeratorLo < product.lo ? 1u : 0u);
        if (residualHi == 0u && residualLo < (bb.significand << 5)) {
            // R < 32*d. Five exact restoring bits replace corrective division.
            uint exactRemainder = residualLo;
            uint digit = 0u;
            const uint threshold4 = bb.significand << 4;
            const uint takeMask4 = exactRemainder >= threshold4 ? 0xffffffffu : 0u;
            exactRemainder -= threshold4 & takeMask4;
            digit |= 16u & takeMask4;
            const uint threshold3 = bb.significand << 3;
            const uint takeMask3 = exactRemainder >= threshold3 ? 0xffffffffu : 0u;
            exactRemainder -= threshold3 & takeMask3;
            digit |= 8u & takeMask3;
            const uint threshold2 = bb.significand << 2;
            const uint takeMask2 = exactRemainder >= threshold2 ? 0xffffffffu : 0u;
            exactRemainder -= threshold2 & takeMask2;
            digit |= 4u & takeMask2;
            const uint threshold1 = bb.significand << 1;
            const uint takeMask1 = exactRemainder >= threshold1 ? 0xffffffffu : 0u;
            exactRemainder -= threshold1 & takeMask1;
            digit |= 2u & takeMask1;
            const uint takeMask0 = exactRemainder >= bb.significand ? 0xffffffffu : 0u;
            exactRemainder -= bb.significand & takeMask0;
            digit |= 1u & takeMask0;
            const uint exactQuotient = guess + digit;
            return PB32Round(sign, exponent, exactQuotient | (exactRemainder != 0u ? 1u : 0u));
        }
    }
#endif
